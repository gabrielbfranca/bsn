           /**
           * Does exactly what roscpp's own default SIGINT handler does: flips
           * ros::ok(   ) so every `while(  ros::ok(   ) )` loop (  run(   )'s,   and the inner one
           * Enactor::body(   ) has,   and ros::spin(   ) inside Logger/ParamAdapter ) ends,  
           * tearDown(   ) runs and main(   ) returns normally.
           *
           * Without it a node built on ROSComponent is simply killed by the signal,  
           * because ros::init(   ) below asks for NoSigintHandler and nothing ever
           * installed a replacement - the target_system side has had its own handler
           * for a long time (  see target_system/Component.cpp's sigIntHandler,   which
           * still overrides this one,   since it is installed later,   in setUp(   ) ),   the
           * management side had none.
           *
           * Two consequences,   both fixed by exiting normally instead:
           * - the node never got a chance to run tearDown(   ) on shutdown;
           * - under a coverage build nothing was ever written: llvm's profile is
           * flushed at exit,   so every node here (  enactor,   data_access,   logger,  
           * param_adapter,   engines,   patient ) reported 0% in coverage.sh's report
           * even though its rostests ran and passed.
           *
           * SIGTERM is handled too because that is what roslaunch escalates to when
           * a node does not exit quickly enough after the SIGINT.
           */
      29   void shutdownOnSignal(  int /*signal*/ ) {
           ros::requestShutdown(   );
           }
          }

           // Crescent indica o quão deslocado um elemento está do começo
           // inverse indica o quão deslocado um elemento está do fim

           double lb = range.getLowerBound(   );
           double ub = range.getUpperBound(   );
           double result;
          
           if(  logic == "decrescent" ) {
           result = fabs(  number - ub );
           result /= (  ub - lb );
           }
           else if(  logic == "crescent" ) {
           result = number - lb;
           result /= (  ub - lb );
           }
           else if(  logic == "medium" ) {
           double mediumValue = (  ub + lb )/2.0;
           result = fabs(  number - mediumValue );
           result /= (  ub - mediumValue );
          
           }
           else {
           throw std::invalid_argument(  "Invalid operation given to getDisplacement" );
           result = -1;
           }
          
           return result;
           }

           int32_t count = 0;
           average = 0;
           int32_t index = 0;
           double bpr_avg = 0.0;
          
           std::vector<double> values;
           std::vector<double>::iterator packets_it;
           for(  packets_it = packetsReceived.begin(   );packets_it != packetsReceived.end(   );++packets_it ){
           if(  static_cast<int>(  *packets_it ) >= 0 ) {
           // Soma à média e retira da fila
           if(  index == 3 || index == 4 ) {
           bpr_avg += *packets_it;
           } else {
           average += *packets_it;
           values.push_back(  *packets_it );
           }
          
           count++;
           }
          
           if(  index == 4 ) {
           if(  bpr_avg >= 0.0 ) {
           bpr_avg /= 2;
           average += bpr_avg;
           values.push_back(  bpr_avg );
           }
           }
          
           index++;
           }
          
           if(  count == 0 )
           return -1;
          
           // Calcula a media partir da soma dividida pelo número de pacotes lidos
           double avg = (  average / count );
          
           std::vector<double> deviations;
           double min = 1000; //Maior valor possível é 100
           double max = -1; //Menor valor possível é 0
          
           size_t i;
          
           for(  i = 0;i < values.size(   );i++ ) {
           //Cálculo dos desvios individuais
           double dev;
           dev = values.at(  i ) - avg;
          
           deviations.push_back(  dev );
          
           if(  dev > max ) {
           max = dev;
           }
          
           if (  dev < min ) {
           min = dev;
           }
           }
          
           double weighted_average = 0.0;
           double weight_sum = 0.0;
          
           // Status de risco do paciente dado em porcentagem
           if(  max - min > 0.0 ) {
           //Se o máximo e mínimo forem diferentes,   normalizar desvios e calcular média ponderada
           for(  i = 0;i < deviations.size(   );i++ ) {
           //Normalizando desvios entre 0 e 1
           deviations.at(  i ) = (  deviations.at(  i ) - min )/(  max - min );
          
           weight_sum += deviations.at(  i );
           weighted_average += values.at(  i )*deviations.at(  i );
           }
          
           risk_status = weighted_average/weight_sum;
           } else {
           //Se o máximo é igual ao mínimo,   a média será calculada e dará o mesmo valor
           risk_status = avg;
           }
